async function process() { const empId = document.getElementById("empId").value; if (!empId) { alert("กรุณากรอกรหัสพนักงาน"); return; } const btn = document.getElementById("submitBtn"); btn.innerText = "กำลังส่ง..."; btn.disabled = true; navigator.geolocation.getCurrentPosition(async (pos) => { const payload = { employeeId: "MT" + empId, latitude: pos.coords.latitude, longitude: pos.coords.longitude, time: new Date().toLocaleString("th-TH"), source: "LIFF" }; try { const res = await fetch(WEBHOOK_URL, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify(payload) }); if (!res.ok) { throw new Error("HTTP " + res.status); } Swal.fire({ icon: "success", title: "ตอกบัตรสำเร็จ", timer: 1500, showConfirmButton: false }).then(() => { liff.closeWindow(); }); } catch (err) { Swal.fire({ icon: "error", title: "ส่งข้อมูลไม่สำเร็จ", text: err.message }); btn.innerText = "กดเพื่อตอกบัตร"; btn.disabled = false; } }, () => { Swal.fire({ icon: "warning", title: "กรุณาเปิด GPS" }); btn.innerText = "กดเพื่อตอกบัตร"; btn.disabled = false; }, { enableHighAccuracy: true }); }
